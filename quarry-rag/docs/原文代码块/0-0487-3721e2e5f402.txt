[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

$resp = Invoke-WebRequest http://127.0.0.1:8100/v1/collections `
  -Headers @{ Authorization = "Bearer $env:QUARRY_KEY" } `
  -UseBasicParsing

$text = [System.Text.Encoding]::UTF8.GetString($resp.RawContentStream.ToArray())
$text | ConvertFrom-Json | ConvertTo-Json -Depth 5
