import os
import sys
import asyncio

if sys.platform.startswith("win"):
    asyncio.set_event_loop_policy(asyncio.WindowsSelectorEventLoopPolicy())

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

# ★ 从 app.models 顶层 import 会触发 models/__init__.py，把四张表都注册进 Base.metadata
from app.models import User, Conversation, Message, ItineraryPlan  # noqa: F401
from app.models.base import init_db

async def main():
    print("[init_db] 开始建表...")
    await init_db()
    print("[init_db] 完成：user, conversation, message, itinerary_plans")

if __name__ == "__main__":
    asyncio.run(main())
