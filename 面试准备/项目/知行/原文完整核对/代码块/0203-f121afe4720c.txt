"""_query_one_leg 演示：查一段班次；查询异常时降级为空段，绝不抛出（纯逻辑，无需密钥）"""
import asyncio

from app.capabilities.train_multi_segment.train_agent_tool import _query_one_leg
from app.capabilities.train_multi_segment.train_skill_schema import TrainLeg

RAW = {
    ("北京", "天津"): """G1 北京南(telecode:VNP) -> 天津(telecode:TJP) 08:30 -> 09:05 历时：00:35
- 二等座: 有票 54.5元
C2001 北京南(telecode:VNP) -> 天津(telecode:TJP) 06:30 -> 07:03 历时：00:33
- 二等座: 无 54.5元""",
    ("天津", "上海"): """G101 天津西(telecode:TXP) -> 上海虹桥(telecode:AOH) 09:10 -> 14:02 历时：04:52
- 二等座: 有 509.0元
G103 天津南(telecode:TIP) -> 上海虹桥(telecode:AOH) 07:48 -> 12:31 历时：04:43
- 二等座: 候补 509.0元""",
}

class FakeTicketTool:        # 测试驱动：假装是 12306 查询工具，按段返回原始文本，并记下被调了几次
    name = "get-tickets"

    def __init__(self):
        self.calls = 0

    async def ainvoke(self, args: dict):
        self.calls += 1
        if args["fromStation"] == "故障站":
            raise RuntimeError("12306 超时")
        return RAW.get((args["fromStation"], args["toStation"]), "")

async def main():
    tool = FakeTicketTool()
    ok = await _query_one_leg(tool, "s1", TrainLeg(from_="北京", to="天津", travel_date="2026-10-01"), "二等座", 5)
    print("正常段：", ok.segment_id, ok.from_, "→", ok.to, [o.option_id for o in ok.train_list])
    bad = await _query_one_leg(tool, "s2", TrainLeg(from_="故障站", to="上海", travel_date="2026-10-02"), "二等座", 5)
    print("异常段：", bad.segment_id, bad.from_, "→", bad.to, "train_list =", bad.train_list)

asyncio.run(main())
