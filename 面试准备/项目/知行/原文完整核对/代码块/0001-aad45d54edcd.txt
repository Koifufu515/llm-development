# 1) 先建本机数据目录（必须为空，路径不要有中文或空格）
mkdir E:\docker-data\postgres

# 2) 运行容器（PowerShell 里换行用反引号 `）
docker run -d `
  --name travel_postgres `
  -e POSTGRES_DB=travel_planner_db `
  -e POSTGRES_USER=travel_user `
  -e POSTGRES_PASSWORD=travel123456 `
  -p 5432:5432 `
  -v E:\docker-data\postgres:/var/lib/postgresql/data `
  --restart unless-stopped `
  pgvector/pgvector:pg17
