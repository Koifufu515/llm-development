"""
Handoffs 主 Agent（本章版本）

本章目标：只跑通"需求收集"这一步。
之后：
- 第 04 章：加入 create_step_config_middleware，Prompt / 工具集按步骤动态注入
- 第 05 章：加入回退工具、记忆工具
- 第 06 章：把 MCP 工具接进来（天气、搜索、酒店、机票...）
- 第 08 章：加入 capabilities 子系统（交通/住宿多段并行）
"""
from langchain.agents import create_agent

from app.core.state import TravelState
from app.core.checkpointer import get_checkpointer
from app.core.store import get_store
from app.utils.llm_factory import get_qwen_llm
from app.agents.handoffs.step_prompts import REQUIREMENT_COLLECTION_PROMPT
from app.tools.state_transition import record_requirement_tool
from app.utils.logger import app_logger

async def create_travel_agent():
    """
    创建"需求收集"版旅行规划 Agent

    返回：
        编译好的 Agent（可直接 ainvoke / astream）

    注意：
        必须在 lifespan 内（或事先手动初始化 Checkpointer / Store 之后）调用，
        因为它会去拿全局的 Checkpointer / Store 实例。
    """
    app_logger.info("创建 Travel Agent（本章版：单步骤）...")

    # 1. 拿大脑
    llm = get_qwen_llm(streaming=True)

    # 2. 拿手脚（本章只有一把）
    all_tools = [record_requirement_tool]

    # 3. 拿持久化
    checkpointer = await get_checkpointer()
    store = await get_store()

    # 4. 组装
    agent = create_agent(
        model=llm,
        tools=all_tools,
        system_prompt=REQUIREMENT_COLLECTION_PROMPT,   # 本章 Prompt 常驻；第 04 章改为中间件动态注入
        state_schema=TravelState,               # 记忆形状（第 02 章定义）
        checkpointer=checkpointer,              # 会话持久化（第 02 章搭好）
        store=store,                            # 长期记忆（第 02 章搭好；第 05 章开始用）
    )

    app_logger.info("✅ Travel Agent 创建完成")
    return agent
