def _clean_keyword(name: str) -> str:
    """ 清洗模型提取的景点名：去序号前缀（"1."/"2、"）与括号注解（"（返程）"），
    提升高德 POI 关键词命中率（原名如"1.杭州东站（返程）"会直接搜空）"""
    name = re.sub(r"^\s*\d+\s*[.、．:：]\s*", "", name)
    name = re.sub(r"[（(][^（）()]*[)）]", "", name)
    return name.strip()
