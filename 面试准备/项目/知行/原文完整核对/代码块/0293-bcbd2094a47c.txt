"""_query_one_segment 演示：查到门店走 hotel 模式；查不到、查询失败、没有工具，一律降级 brand 模式（纯逻辑，无需密钥）"""
import asyncio
import json

from app.capabilities.hotel_multi_segment.hotel_agent_tool import _query_one_segment
from app.capabilities.hotel_multi_segment.hotel_skill_schema import HotelSegmentInput

HOTELS = {"杭州": json.dumps({"code": 0, "hotelInformationList": [
    {"name": "如家酒店(西湖北山街店)", "price": {"message": "716CNY 共2晚 358CNY/晚"}, "starRating": 4.6, "address": "杭州市西湖区北山街 12 号"},
    {"name": "锦江之星(Jinjiang Inn)", "price": {"message": "560CNY 共2晚 280CNY/晚"}, "starRating": 4.3, "address": "杭州市拱墅区体育场路 8 号"},
]}, ensure_ascii=False)}

class FakeHotelTool:         # 测试驱动：假装是酒店查询工具（带真实工具那种嵌套 schema），按城市返回结果
    name = "searchHotels"
    args_schema = {"type": "object", "properties": {"originQuery": {}, "place": {}, "checkInParam": {}, "filterOptions": {}, "size": {}}}

    async def ainvoke(self, args: dict):
        if args["place"] == "故障城":
            raise RuntimeError("酒店服务超时")
        return HOTELS.get(args["place"], "")

async def main():
    for i, (tool, city) in enumerate([(FakeHotelTool(), "杭州"), (FakeHotelTool(), "故障城"), (None, "苏州")], 1):
        seg = await _query_one_segment(tool, f"h{i}", HotelSegmentInput(city=city, checkInDate="2026-10-01", checkOutDate="2026-10-03",
                                                                        guestCount=2, priceLevel="经济型"), 5)
        picks = [h.hotel_id + " " + h.hotelName for h in seg.hotel_list] or [b.brand_id + " " + b.brandName for b in seg.brand_list[:3]]
        print(f"{seg.segment_id} {city:3} mode={seg.mode:5} {picks}")

asyncio.run(main())
