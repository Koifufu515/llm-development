"""
MCP 工具筛选器
按业务类别（酒店 / 航班 / 高铁 / 自驾 / 天气 / 搜索 / 日期）过滤 MCP 工具
"""
from typing import List
from langchain_core.tools import BaseTool

from app.mcp_core.client import get_mcp_client
from app.utils.logger import app_logger

# ============== 全量 ==============

async def get_all_mcp_tools() -> List[BaseTool]:
    """获取所有 MCP 工具"""
    manager = await get_mcp_client()
    tools = await manager.get_tools()
    app_logger.info(f"📦 全量 MCP 工具: {len(tools)} 个")
    return tools

# ============== 分类筛选器 ==============

async def get_hotel_tools() -> List[BaseTool]:
    """酒店类工具（aigohotel）"""
    manager = await get_mcp_client()
    all_tools = await manager.get_tools()
    hotel_tools = [
        t for t in all_tools
        if any(k in t.name.lower() for k in ["find-hotels", "searchhotels"])
    ]
    app_logger.info(f"🏨 酒店工具: {[t.name for t in hotel_tools]}")
    return hotel_tools

async def get_flight_tools() -> List[BaseTool]:
    """航班类工具（VariFlight）"""
    manager = await get_mcp_client()

        t for t in all_tools if "flight" in t.name.lower()
