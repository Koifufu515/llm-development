"""_weather_paragraph 演示：代码直查天气并按行程窗口段落式渲染（需 .env 配好天气 MCP / 高德密钥）"""
import asyncio
from datetime import date, timedelta

from app.services.chat_stream_executor import _weather_paragraph

async def main():
    start = (date.today() + timedelta(days=1)).isoformat()
    print(await _weather_paragraph(["杭州"], start, 3) or "（查不到 → 空串，调用方不追加、不编造）")
    print("\n—— 多城 ——")
    print(await _weather_paragraph(["杭州", "上海"], start, 2))

if __name__ == "__main__":
    asyncio.run(main())
