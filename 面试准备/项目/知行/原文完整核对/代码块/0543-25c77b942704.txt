from app.services.chat_contract_service import _manual_multi_city_day_warning

cases = {
    "单目的地": "我想去杭州旅行\n- 旅行天数：2 天",
    "3 城 4 天": "我想去杭州、上海、苏州旅行\n- 旅行天数：4 天",

    "非表单": "帮我看看杭州、上海怎么玩",
