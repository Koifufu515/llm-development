"""第 11 章综合冒烟：不调 LLM，用真实工具把七步前向链走完，再按用户原话回退（纯逻辑，无需密钥）

两个测试驱动函数只做"LangGraph 本来会做的事"：把用户发言塞进 messages、把工具返回的 Command.update
合并回 state（messages 追加，其余字段覆盖）。所有业务判定都在真实工具里。
"""
from langchain.tools import ToolRuntime
from langchain_core.messages import HumanMessage

from app.tools import state_transition as st

state = {"messages": [], "current_step": "requirement_collection"}

def say(text):
    state["messages"].append(HumanMessage(content=text))

def run(tool, **kwargs):
    rt = ToolRuntime(state=state, context=None, config={}, stream_writer=None,
                     tool_call_id=f"call_{len(state['messages'])}", store=None)
    up = tool.func(runtime=rt, **kwargs).update
    state["messages"] += up.get("messages", [])
    state.update({k: v for k, v in up.items() if k != "messages"})
    return up

say("我想去天津、上海旅行\n- 出发地：北京\n- 出发日期：2026-10-01\n- 旅行天数：4 天\n- 出行人数：2 成人")
run(st.record_requirement_tool, departure_city="北京", departure_date="2026-10-01", travel_days=3,
    budget_min=2000, budget_max=3000, travel_styles=["culture"], adult_count=2)
assert state["user_requirement"]["travel_days"] == 4 and state["current_step"] == "destination_recommendation"
print("✅ 1 需求收集：天数以表单为准（4），推进到目的地步")

say("就去天津、上海")
run(st.select_destination_tool, destination="天津、上海")
assert state["current_step"] == "transport_planning" and st._expected_legs(state) == ["北京→天津", "天津→上海"]
print("✅ 2 目的地：写入完整多城串，应有 2 段交通")

say("北京到天津坐 G1")
run(st.select_transport_tool, transport_type="train",
    legs=[{"from_city": "北京", "to_city": "天津", "vehicle_no": "G1", "depart_time": "08:00", "price": 54.5}])
assert state["leg_cursor"] == 1 and state["current_step"] == "transport_planning"
say("天津到上海坐 G7512")
run(st.select_transport_tool, transport_type="train",
    legs=[{"from_city": "天津", "to_city": "上海", "vehicle_no": "G7512", "depart_time": "09:05", "price": 553}])
assert state["current_step"] == "accommodation_planning" and state["leg_prices"] == {"北京→天津": 54.5, "天津→上海": 553.0}
print("✅ 3 交通：段级循环两轮，最后一段才推进；两段票价同步进数字账本")

say("天津住如家")
run(st.select_accommodation_tool, accommodation_types=["economy_hotel"], hotels=["天津 如家酒店 199元/晚"])
say("上海住汉庭")
run(st.select_accommodation_tool, accommodation_types=["economy_hotel"], hotels=["上海 汉庭酒店 259元/晚"])
assert state["current_step"] == "food_planning" and len(state["selected_hotels"]) == 2
print("✅ 4 住宿：逐城循环两轮，推进到餐饮步")

say("本地小吃就行")
run(st.select_food_tool, food_types=["local"])
assert state["current_step"] == "itinerary_generation"
print("✅ 5 餐饮：推进到行程步")

run(st.submit_itinerary_body_tool, days=[{"day": 1, "city": "天津", "morning": ["五大道骑行，建议停留 2 小时，沿途拍照"]}])
assert len(state["itinerary_body"]) == 4 and state.get("budget")
print("✅ 6 行程：只交 1 天，代码归一补齐到 4 天，并就地补算预算")

run(st.summarize_budget_tool, transport_price_per_person=560, hotel_price_per_night=229, travel_days=4, budget_max=3500)
assert state["user_requirement"]["budget_max"] == 3500 and state["current_step"] == "budget_summarization"
print("✅ 7 预算：参数覆盖并写回需求真源，流程停在终点")

run(st.go_back_to_destination)
assert state["selected_destination"] == "天津、上海"
print("✅ 8 回退守卫：用户没说要换，快捷回退被拒，数据原样保留")

say("换个目的地吧")
run(st.go_back_to_destination)
assert state["current_step"] == "destination_recommendation"
assert state["selected_destination"] is None and state["leg_cursor"] is None and state["stay_cursor"] is None
assert state["user_requirement"]["travel_days"] == 4
print("✅ 9 回退：目的地及之后全部清空（含两个游标），需求保留")
print("\n🎉 第 11 章综合冒烟全部通过")
