"""_trim_window 演示：消息超过 40 条才裁，刀口只落在用户消息上，整轮丢最旧的，当前轮永不裁（纯函数，无需密钥）"""
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage, ToolMessage

from app.core.middleware import _trim_window

msgs = [SystemMessage("你是旅行助手")]
for turn in range(1, 12):                                     # 11 轮，每轮 用户 + 调工具 + 工具结果 + 回复 = 4 条
    msgs += [HumanMessage(f"第{turn}轮提问"), AIMessage("", tool_calls=[{"name": "q", "args": {}, "id": f"c{turn}"}]),
             ToolMessage("结果", tool_call_id=f"c{turn}"), AIMessage(f"第{turn}轮回答")]
out = _trim_window(msgs)
print(f"裁剪前 {len(msgs)} 条 → 裁剪后 {len(out)} 条")
print("保留系统消息：", out[0].type == "system", "| 第一条对话：", out[1].type, out[1].content)
print("20 条以内不动：", _trim_window(msgs[:20]) is msgs[:20] or len(_trim_window(msgs[:20])) == 20)
