"""高铁数据契约演示：字段别名、票价序列化、按别名导出给前端（纯逻辑，无需密钥）"""
from app.capabilities.train_multi_segment.train_skill_schema import TrainLeg, TrainMultiSegmentPayload, TrainOption, TrainSegment

leg = TrainLeg.model_validate({"from": "北京", "to": "天津", "date": "2026-10-01"})    # 前端/模型可以用 from、date
print("入参别名：", leg.from_, leg.to, leg.travel_date)
opt = TrainOption(trainNo="G1", startTime="08:30", endTime="09:05", startStation="北京南", endStation="天津", duration="00:35",
                  price=54.0, seatType="二等座", ticketStatus="有票", option_id="G1-08:30-天津")
seg = TrainSegment(segment_id="s1", from_="北京", to="天津", date="2026-10-01", train_list=[opt])
out = TrainMultiSegmentPayload(segments=[seg]).model_dump(by_alias=True)
print("下发给前端：", out["skill_name"], "| 段字段：", list(out["segments"][0]), "| 票价：", out["segments"][0]["train_list"][0]["price"])
