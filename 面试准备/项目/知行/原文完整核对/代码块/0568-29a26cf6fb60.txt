"""_render_ticket_table 演示：普速过滤、按段筛选、面向用户的表头（纯函数，无需密钥）"""
from app.services.chat_stream_executor import _render_ticket_table, _rows_to_seg

seg1 = _rows_to_seg([
    {"code": "G1650", "dep": "08:12", "arr": "12:30", "dur": "4:18", "seat": "二等座 有 356元"},
    {"code": "K1086", "dep": "06:30", "arr": "19:45", "dur": "13:15", "seat": "硬座 有 128元"},   # 段内有 G → 普速被剔
])
seg2 = _rows_to_seg([{"code": "G7512", "dep": "09:05", "arr": "10:02", "dur": "0:57", "seat": "二等座 有 73元"}])
segs = {("福州", "杭州东"): seg1, ("杭州东", "上海虹桥"): seg2}

print(_render_ticket_table(segs))                               # 默认表头：写给模型看的自约束
print(_render_ticket_table(segs, only_leg="杭州→上海"))          # 多目的地段级循环：只出当前段
print(_render_ticket_table(segs, header="**12306 实时班次**\n", tail="\n以上为查询结果。\n")[:60], "…")
