def plan_to_text(plan_data: dict) -> str:
    """结构化行程 JSON → 裁判可读的行程正文（与模型输出形态对齐）"""

                f"{s.get('toponym', '')}（{str(s.get('toponym_desc', ''))[:60]}）"
                for s in spots
            ]
            lines.append(f"{item.get('time', '')}：{'、'.join(parts)}")
    return "\n".join(lines)

async def run_quality_check(plan_data: dict, threshold: float = 0.5) -> None:
    """对单份落库行程做三维度质检：通过打 INFO，不通过打 WARNING + 裁判理由

    设计为"事后告警"：行程已落库发给用户，质检不重生成、不阻断；
    异常全吞（fail-safe），绝不影响主对话流程。
    """
    title = plan_data.get("title", "未命名行程")
    try:
        case = LLMTestCase(
            input=title,  # title 由需求生成，是最忠实的需求快照
            actual_output=plan_to_text(plan_data),
        )
        scores: dict[str, float] = {}
        reasons: list[str] = []
        all_reasons: list[str] = []
        # 在线质检的源料是 plan_to_text 产出的结构化清单（无预算字段），
        # 故 require_budget=False——否则"完整性"维度会因恒缺预算而系统性扣分、刷出噪音告警
        for metric in build_itinerary_metrics(threshold=threshold, require_budget=False):  # ★ 在线源料无预算
            await metric.a_measure(case)
            scores[metric.name] = metric.score
            # 所有维度的裁判理由都收集，通过时也记录（便于观察分数波动原因）
            all_reasons.append(f"{metric.name}: {metric.reason}")
            if metric.score < threshold:
                reasons.append(f"{metric.name}: {metric.reason}")

        low = {k: v for k, v in scores.items() if v < threshold}
        if low:
            app_logger.warning(
                f"⚠️ 行程质检未达标: {title} | 分数: {scores} | 问题: {'；'.join(reasons)}"
            )
        else:
            # 通过时同样带裁判理由（截断防日志爆炸），低分维度可追溯
            brief = "；".join(all_reasons)
            if len(brief) > 500:
                brief = brief[:500] + "……"
            app_logger.info(f"✅ 行程质检通过: {title} | 分数: {scores} | 理由: {brief}")
    except Exception as e:
        # 质检自身失败（网络/限流/结构化解析）只记录，不影响业务
        app_logger.warning(f"行程质检执行异常（已忽略）: {title} | {type(e).__name__}: {e}")

def schedule_quality_check(plan_data: dict) -> None:
    """落库后调用：创建后台质检任务（不阻塞回复流）"""
    task = asyncio.create_task(run_quality_check(plan_data))  # ★ 起后台任务，立刻返回
    _BG_TASKS.add(task)
    task.add_done_callback(_BG_TASKS.discard)

"""
意图路由正确性指标，基于DeepEval自定义Metric
路由输出是强类型RouterResult，包含planning布尔值和skills列表，属于分类任务
直接用断言打分：对得1分，错得0分，不需要调用LLM做裁判
- planning是核心开关，必须和预期完全一致
- skills分3种校验逻辑：
  expect_empty_skills=True → skills必须是空
  expected_skills_any不为空 → 实际skills只要包含里面任意一个就算通过
  两个参数都不传 → 不校验skills
"""
import json
from deepeval.metrics import BaseMetric
from deepeval.test_case import LLMTestCase

class RouterCorrectnessMetric(BaseMetric):
    """意图路由正确性校验：planning精确匹配 + skills多模式校验"""
    def __init__(
        self,
        expected_planning: bool,
        expected_skills_any: list[str] | None = None,
        expect_empty_skills: bool = False,
    ):
        # 两个skills参数不能同时开启，逻辑冲突，直接抛异常，不静默忽略配置错误
        if expect_empty_skills and expected_skills_any:
            raise ValueError("expect_empty_skills=True 和 expected_skills_any 非空不能同时设置，二选一")

        # 预期的规划分流结果，这个是必校验项
        self.expected_planning = expected_planning
        # 期望匹配的技能列表，None/空代表不校验skills
        self.expected_skills_any = expected_skills_any or []
        # 标记：是否要求skills必须为空，用来校验闲聊场景
        self.expect_empty_skills = expect_empty_skills

        # 阈值固定1.0，必须在构造函数赋值，防止外部赋值时机不对导致is_successful异常
        self.threshold = 1.0
        self.score = 0.0
        self.reason = ""

    def measure(self, test_case: LLMTestCase) -> float:
        # actual_output是测试用例传过来的RouterResult序列化后的json字符串
        try:
            actual = json.loads(test_case.actual_output)
        except (json.JSONDecodeError, TypeError):
            self.score, self.reason = 0.0, "路由返回不是合法JSON，结构化输出失败"
            return self.score

        # 取出planning标记，转布尔
        actual_planning = bool(actual.get("is_itinerary_planning"))
        # 提取所有技能名称，组装成列表
        actual_skills = [s.get("skill_name") for s in (actual.get("skills") or [])]

        # 第一步校验：planning必须完全匹配，这是最容易出bug的地方
        if actual_planning != self.expected_planning:
            self.score = 0.0
            self.reason = f"planning期望 {self.expected_planning}，实际 {actual_planning}"
            return self.score

        # 第二步校验skills，分三种情况
        # 场景1：闲聊类，要求技能列表必须是空
        if self.expect_empty_skills:
            if actual_skills:
                self.score = 0.0
                self.reason = f"期望技能为空，实际拿到 {actual_skills}"
                return self.score
            self.score, self.reason = 1.0, "planning正确，技能列表为空"
            return self.score

        # 场景2：技能宽松匹配，只要命中任意一个目标技能就通过
        if self.expected_skills_any:
            hit = [s for s in actual_skills if s in self.expected_skills_any]
            if not hit:
                self.score = 0.0
                self.reason = (
                    f"skills需要命中 {self.expected_skills_any} 其中一个，实际 {actual_skills}"
                )
                return self.score
            self.score, self.reason = 1.0, f"planning正确，命中技能 {hit}"
            return self.score

        # 场景3：不校验skills，planning正确就直接满分
        self.score, self.reason = 1.0, f"planning正确（skills={actual_skills}）"
        return self.score

    async def a_measure(self, test_case: LLMTestCase) -> float:
        # 异步版本直接复用同步逻辑，不用单独写
        return self.measure(test_case)

    def is_successful(self) -> bool:
        # 判断指标是否达标：分数大于等于阈值
        return self.score >= self.threshold

"""
意图路由评测入口
两种运行方式任选：
- uv run pytest evaluation/test_router.py -v        # 原生pytest执行
- deepeval test run evaluation/test_router.py      # DeepEval CLI，结果可以上传Confident平台

逻辑：遍历所有测试用例，调用真实的 chat_route 路由函数。
把返回的强类型RouterResult转成JSON字符串，交给自定义指标做固定规则校验。
"""
import asyncio
import json
import os

# Windows环境下异步客户端需要Selector事件循环，和项目app/run.py保持一致
if os.name == "nt":
    asyncio.set_event_loop_policy(asyncio.WindowsSelectorEventLoopPolicy())

from deepeval import assert_test
from deepeval.test_case import LLMTestCase
import pytest
from app.agents.routers.chat_router import chat_route
from evaluation.router_cases import CASES, RouterCase
from evaluation.router_metric import RouterCorrectnessMetric

@pytest.mark.parametrize("case", CASES, ids=[c.note for c in CASES])
def test_chat_router(case: RouterCase):
    """单条路由测试：调用真实路由，校验planning和skills结果"""
    # 执行路由函数，async方法用asyncio.run转为同步调用；history用于对话上下文接续
    result = asyncio.run(chat_route(case.msg, history=case.history or None))

    # 将路由返回对象序列化JSON，作为deepeval测试的实际输出
    actual_output = json.dumps({
        "is_itinerary_planning": result.is_itinerary_planning,
        "skills": [s.model_dump() for s in result.skills],
    }, ensure_ascii=False)

    test_case = LLMTestCase(
        input=case.msg,
        actual_output=actual_output,
    )

    # 初始化自定义校验指标，必须全部匹配才能通过
    metric = RouterCorrectnessMetric(
        expected_planning=case.planning,
        expected_skills_any=case.skills,
        expect_empty_skills=case.expect_empty_skills,
    )
    metric.threshold = 1.0
    assert_test(test_case, [metric])
