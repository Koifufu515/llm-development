async def create_travel_agent():
    """
    创建"需求收集"版旅行规划 Agent

    返回：
        编译好的 Agent（可直接 ainvoke / astream）

def _print_ai_reply(reply):
    """把 AIMessage 的可读部分打出来"""
    if isinstance(reply.content, str):
        text = reply.content
    else:
        # 部分模型返回 list[ContentBlock] 结构
        text = "".join(
            b.get("text", "") for b in reply.content if isinstance(b, dict)
        )
    print(f"     Agent 回复: {text[:200]}{'...' if len(text) > 200 else ''}")

def _print_tool_call(reply):
    """如果这一轮 LLM 调了工具，打印工具名和参数"""
    if not reply.tool_calls:
        return
    for tc in reply.tool_calls:
        print(f"     🔧 调用工具: {tc['name']}")
        print(f"        参数: {tc['args']}")

async def run_conversation():
    """跑完四轮对话"""
    print("\n" + "=" * 60)
    print(" 阶段 1  开始四轮对话")
    print("=" * 60)

    agent = await create_travel_agent()
    config = {
        "configurable": {
            "thread_id": THREAD_ID,
            "user_id": USER_ID,
        }
    }

    for i, user_text in enumerate(USER_TURNS, start=1):
        print(f"\n  [第 {i} 轮]")
        print(f"     用户: {user_text}")

        # 每一轮只往图里推**用户这一条新消息**
        # Checkpointer 会自动带回历史，无需手动拼
        input_state = {"messages": [HumanMessage(content=user_text)]}

        result = await agent.ainvoke(input_state, config=config)

        # 最后一条 message 一般是 AIMessage
        last_msg = result["messages"][-1]

        if isinstance(last_msg, AIMessage):
            _print_ai_reply(last_msg)
            _print_tool_call(last_msg)
        elif isinstance(last_msg, ToolMessage):
            # 工具刚执行完，图会继续回到 Model 节点再吐一条 AIMessage
            print(f"     🛠️ 工具返回: {last_msg.content[:200]}")

        # 打印 State 里的 current_step / user_requirement 变化
        cs = result.get("current_step")
        ur = result.get("user_requirement")
        if cs:
            print(f"     📌 current_step = {cs}")
        if ur:
            print(f"     📌 user_requirement = {ur}")

    """阶段 2：从 Checkpointer 读回最终 State，核对字段"""
    print("\n" + "=" * 60)
    print(" 阶段 2  从 Checkpointer 核对最终 State")
    print("=" * 60)

    agent = await create_travel_agent()
    config = {"configurable": {"thread_id": THREAD_ID}}

    # 直接从 Agent 的 State 拉当前值
    state = await agent.aget_state(config)

    values = state.values
    ur = values.get("user_requirement")
    cs = values.get("current_step")
    msgs = values.get("messages", [])

    print(f"  📖 current_step: {cs}")
    print(f"  📖 消息总数: {len(msgs)}")
    if ur:
        print(f"  📖 user_requirement 详情:")
        for k, v in ur.items():
            print(f"     - {k}: {v}")

    # 校验：跑完对话后 user_requirement 应当已就绪
    if ur is None:
        print("  ⚠️  user_requirement 还没填上——可能是模型没决定调工具")
        print("      重跑一次或改用更明确的第 3 轮说法（比如把预算 & 风格写清楚）")
        return False

    assert ur.get("departure_city") == "北京", "❌ 出发地不对"
    assert ur.get("travel_days") == 5, "❌ 天数不对"
    assert "culture" in (ur.get("travel_styles") or []) or \
           "food" in (ur.get("travel_styles") or []), "❌ 风格未记录"
    print("\n  ✅ 关键字段核对通过")

@tool
def record_requirement_tool(
        departure_city: str,
        departure_date: str,
        travel_days: int,
        budget_min: float,
        budget_max: float,
        travel_styles: list[str],
        special_needs: str = "",
        adult_count: Optional[int] = 1,
        children_count: Optional[int] = 0,
        destination: Optional[str] = None,
        runtime: ToolRuntime[None, TravelState] = None
) -> Command:
    """记录用户旅行需求，并进入目的地推荐步骤。
    (docstring 与第 03 章一致，此处压缩)
    """
    # ↓↓↓ 与第 03 章完全一致，不再重复注释 ↓↓↓
    try:
        datetime.strptime(departure_date, "%Y-%m-%d")
    except ValueError:
        return Command(update={"messages": [ToolMessage(
            content="❌ 日期格式错误，请使用 YYYY-MM-DD 格式",
            tool_call_id=runtime.tool_call_id
        )]})

    avg_budget = (budget_min + budget_max) / 2
    if avg_budget < 3000:
        budget_level = "economy"
    elif avg_budget < 8000:
        budget_level = "comfort"
    else:
        budget_level = "luxury"

    requirement = UserRequirement(
        departure_city=departure_city,
        destination=destination,
        departure_date=departure_date,
        travel_days=travel_days,
        adult_count=adult_count,
        children_count=children_count,
        budget_min=budget_min,
        budget_max=budget_max,
        budget_level=budget_level,
        travel_styles=travel_styles,
        special_needs=special_needs if special_needs else None,
    )
    app_logger.info(f"[STEP] 需求已记录 → destination_recommendation")
    return Command(update={
        "messages": [ToolMessage(
            content=(f"✅ 需求已记录！\n出发：{departure_city} → 目的地：{destination or '待推荐'}\n"
                     f"{travel_days} 天 | 预算 {budget_min}-{budget_max} 元/人（{budget_level}）"),
            tool_call_id=runtime.tool_call_id
        )],
        "user_requirement": requirement,
        "current_step": "destination_recommendation",
    })

@tool
def select_accommodation_tool(

    确认用户选择的住宿类型，进入餐饮规划步骤。

    参数：

    return Command(update={

            tool_call_id=runtime.tool_call_id

@tool
def generate_itinerary_tool(
        activities_by_day: list[list[str]],
        runtime: ToolRuntime[None, TravelState] = None
) -> Command:
    """
    落库逐日行程，进入预算汇总步骤。

    参数：
    - activities_by_day: 二维列表，外层长度 = 天数，
      内层每一项是该天的活动列表。
      例如 3 天：
        [
          ["黄鹤楼", "汉口江滩", "户部巷小吃"],
          ["东湖磨山", "湖北省博物馆"],
          ["武汉大学", "光谷步行街"]
        ]

    调用时机：
    - 用户对目的地、交通、住宿、餐饮全部确认完毕
    - 模型基于选定的目的地和天数拟出每日活动后
    """
    state = runtime.state
    req = state.get("user_requirement") or {}
    travel_days = int(req.get("travel_days") or 0)
    destination = state.get("selected_destination") or req.get("destination") or ""

    # 校验：activities_by_day 长度必须与旅行天数匹配
    if len(activities_by_day) != travel_days:
        return Command(update={"messages": [ToolMessage(
            content=(f"❌ 行程天数不匹配：需求是 {travel_days} 天，"

            tool_call_id=runtime.tool_call_id
        )]})

    # 构造结构化行程（ItineraryDay 是第 02 章定义的 TypedDict）
    itinerary: list[ItineraryDay] = []

        itinerary.append(ItineraryDay(
            day_number=i,
            activities=day_activities,
            meals=[],           # 骨架版不细化到每餐，第 08 章交给 capabilities 出
            accommodation="",   # 骨架版不细化到每晚住哪，同上
        ))

    app_logger.info(
        f"[STEP] 行程已生成: {destination} · {travel_days} 天 → budget_summarization"
    )

    return Command(update={
        "messages": [ToolMessage(
            content=f"✅ 已生成 {travel_days} 天行程，共 {sum(len(x) for x in activities_by_day)} 项活动",
            tool_call_id=runtime.tool_call_id
        )],
        "itinerary": itinerary,
        "current_step": "budget_summarization",
    })

# ============== 7️⃣ 预算汇总工具（骨架版） ==============

@tool
def summarize_budget_tool(
        transport_cost: float,     # 交通（元/人/全程）
        accommodation_cost: float, # 住宿（元/人/全程）
        food_cost: float,          # 餐饮（元/人/全程）
        attractions_cost: float,   # 景点门票（元/人/全程）
        misc_cost: float = 0.0,    # 其他（元/人/全程）
        runtime: ToolRuntime[None, TravelState] = None
) -> Command:
    """
    汇总预算，进入报告生成步骤（最终步）。

    参数说明（全部按"每人全程"口径填）：
    - transport_cost: 交通费（含所有城际段）
    - accommodation_cost: 住宿费（全部晚数合计）
    - food_cost: 餐饮费
    - attractions_cost: 景点门票
    - misc_cost: 其他（保险、装备等，可缺省）

    调用时机：
    - 行程已经生成（itinerary 字段非空）
    - 模型基于选定的目的地/交通/住宿/餐饮档位估算出各项费用后
    """
    total = (transport_cost + accommodation_cost +
             food_cost + attractions_cost + misc_cost)

    budget = BudgetBreakdown(
        transport=transport_cost,
        accommodation=accommodation_cost,
        food=food_cost,
        attractions=attractions_cost,
        misc=misc_cost,
        total=total,
    )

    # 与需求预算做个简单对比
    req = runtime.state.get("user_requirement") or {}
    b_min = float(req.get("budget_min") or 0)
    b_max = float(req.get("budget_max") or 0)
    overrun_note = ""
    if b_max and total > b_max:
        overrun_note = f"（超出预算上限 {total - b_max:.0f} 元）"
    elif b_min and total < b_min:
        overrun_note = f"（低于预算下限 {b_min - total:.0f} 元）"

    app_logger.info(
        f"[STEP] 预算已汇总: {total:.0f} 元/人 {overrun_note} → report_generation"
    )

    return Command(update={
        "messages": [ToolMessage(
            content=(f"✅ 预算已汇总（人均全程）：\n"
                     f"  交通 {transport_cost:.0f} + 住宿 {accommodation_cost:.0f} + "
                     f"餐饮 {food_cost:.0f} + 景点 {attractions_cost:.0f} + "
                     f"其他 {misc_cost:.0f}\n"
                     f"  合计 {total:.0f} 元/人 {overrun_note}"),
            tool_call_id=runtime.tool_call_id
        )],
        "budget": budget,
        "current_step": "report_generation",
    })
