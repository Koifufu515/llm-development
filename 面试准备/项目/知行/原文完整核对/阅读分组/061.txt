    """查询单段住宿（由 asyncio.gather 并行调度）

        "priceLevel": segment.priceLevel,
    }

        return HotelMultiSegmentItem(
            **base, mode="hotel", hotel_list=_reid(segment_id, options, "hotel_id"), brand_list=[],
        )

    max_options_per_segment: int = 10,

    """核心函数（非工具）：并行查询全部住宿段 → 组装 payload，并缓存供提交侧校验

    段数由入参长度决定（2/3/多段均支持，不写死）；每段模式独立判定，可混合（h1 门店 + h2 品牌）。

                  for seg in segments]

async def auto_select_hotels(conversation_id: str, state: dict) -> list[str]:

    返回已选定站的展示文本（空列表 = 未推进，调用方跳过即可）
    """
    segments = _segments(state)
    if not segments:
        return []
    payload = await _payload_for(conversation_id, segments)
    texts: list[str] = []
    prices: dict[str, float] = {}
    for seg in payload.segments:
        head = f"{seg.city}（{seg.checkInDate} 至 {seg.checkOutDate}）"
        if seg.hotel_list:
            # ★ 门店择优与快速通道同源（同一批候选）：评分≥4.5 取低价、否则取评分最高，
            #   异常低价剔除；全为异常低价时不硬选，本轮不推进（交回原流程如实告知）
            hotel = pick_best_hotel(list(seg.hotel_list))
            if hotel is None:
                break
            texts.append(
                f"{head}：{hotel.hotelName}，{hotel.pricePerNight:g}元/晚，总价 {hotel.totalPrice:g}元"
            )
            if float(hotel.pricePerNight) > 0:     # 同一次循环产出数字账本
                prices[seg.city] = float(hotel.pricePerNight)
        elif seg.brand_list:
            texts.append(
                f"{head}：{seg.brand_list[0].brandName}（实时报价不可用，仅选定品牌，"
                f"具体门店到店择优选同档位）"
            )
    if len(texts) != len(payload.segments):
        app_logger.warning(
            f"⚠️ 住宿自动模式代选不完整（{len(texts)}/{len(payload.segments)} 站有候选），本轮不推进"
        )
        return []
    agent, config, _ = await get_agent_and_state(conversation_id)
    extra, step = _next_step(state)
    upd: dict = {
        "selected_hotels": texts, "stay_cursor": len(texts), "current_step": step,
        "stay_prices": prices,      # ★ 数字账本：第 11 章预算工具直接读，不解析展示文本
        # 档位必须**无条件写入**（表单未预选时兜底经济型）。
        # 行程步 requires 硬校验 selected_accommodation_types，字段缺失会直接抛错卡死流程
        "selected_accommodation_types": _accom_types(state) or ["economy_hotel"],
        **extra,
    }
    await agent.aupdate_state(config, upd)
    app_logger.info(f"✅ 住宿自动模式已代选 {len(texts)} 站，推进到 {step}")
    return texts

def _repair_dangling_tool_calls(msgs: list) -> tuple[list, int, int]:
    """修复悬空 tool_calls / 孤立 tool 消息
    返回 (处理后的消息列表, 补入的 ToolMessage 条数, 丢弃的孤立 ToolMessage 条数)；
    无需修复时原样返回入参对象（零拷贝，不影响正常路径）
    """
    out: list = []
    inserted = dropped = 0
    i, n = 0, len(msgs)
    while i < n:
        m = msgs[i]
        mtype = getattr(m, "type", None)
        if mtype == "ai" and getattr(m, "tool_calls", None):
            out.append(m)
            # 记录本AI消息的tool_call_id，后续匹配已响应的工具消息
            pending = {
                tc.get("id"): tc.get("name")
                for tc in m.tool_calls if tc.get("id")
            }
            j = i + 1
            while j < n and getattr(msgs[j], "type", None) == "tool":
                pending.pop(getattr(msgs[j], "tool_call_id", None), None)
                out.append(msgs[j])
                j += 1
            # 对未得到响应的tool_call，补入占位ToolMessage
            for tid, tname in pending.items():
                out.append(ToolMessage(
                    content=_TOOL_ABORTED_HINT, tool_call_id=tid, name=tname or "tool"
                ))
                inserted += 1
            i = j
            continue
        # 没有对应AI tool_call的孤立tool消息，直接丢弃
        if mtype == "tool":
            dropped += 1
            i += 1
            continue
        out.append(m)
        i += 1
    if inserted or dropped:
        return out, inserted, dropped
    return msgs, 0, 0

# 仅裁剪发往模型的视图，checkpoint持久化历史完整保留；
# 切点只在HumanMessage边界，不切割AI-Tool消息配对，避免产生残缺工具历史
_WINDOW_TRIGGER = 40   # 消息数超过该阈值才执行裁剪
_WINDOW_KEEP = 24      # 目标最少保留消息条数

def _trim_window(msgs: list) -> list:
    """按用户轮边界丢弃最旧历史，找不到安全切点则原样返回"""
    if len(msgs) <= _WINDOW_TRIGGER:
        return msgs
    last_human = -1
    for i in range(len(msgs) - 1, -1, -1):
        if isinstance(msgs[i], HumanMessage):
            last_human = i
            break
    if last_human <= 0:
        # 无更早可裁剪的用户轮
        return msgs
    # 寻找裁剪切点
    cut = min(len(msgs) - _WINDOW_KEEP, last_human - 1)
    for i in range(cut, -1, -1):
        if isinstance(msgs[i], HumanMessage):
            kept_system = [m for m in msgs[:i] if getattr(m, "type", None) == "system"]
            return kept_system + msgs[i:]
    return msgs

import re
_DAY_MARK_RE = re.compile(r"第\s*(\d+|[一二三四五六七八九十]+)\s*天|Day\s*(\d+)", re.IGNORECASE)
_CN_DIGIT = {"一": 1, "二": 2, "三": 3, "四": 4, "五": 5, "六": 6, "七": 7, "八": 8, "九": 9}

def _day_numbers(text: str) -> set[int]:
    """ 提取正文里出现过的天序号（第3天 / 第三天 / Day 3）"""
    nums: set[int] = set()
    for m in _DAY_MARK_RE.finditer(text or ""):
        if m.group(2):                                  # Day 4 / day 12
            nums.add(int(m.group(2)))
            continue
        g1 = m.group(1)                                 # 第3天 / 第1天 / 第三天
        if g1.isdigit():
            nums.add(int(g1))
            continue
        cn = g1
        if len(cn) == 1:
            nums.add(10 if cn == "十" else _CN_DIGIT.get(cn, 0))
        elif cn.startswith("十"):
            nums.add(10 + _CN_DIGIT.get(cn[1], 0))
        elif cn.endswith("十"):
            nums.add(_CN_DIGIT.get(cn[0], 0) * 10)
    return {n for n in nums if n > 0}

# ============== 段级循环：用户所选班次回填 ==============

# 车次号（G1197/D50/K507/T10/Z162）或航班号（CA4105/MU5138）。
# 结尾用 (?![0-9A-Za-z]) 而非 \b：\b 在 Python 里按 Unicode 判定，中文也算 word 字符，
# "G1197吧"这类用户口语写法会被判成"词内无边界"而漏匹配（实测第一条就没回填上）
_VEHICLE_CODE_RE = re.compile(r"[GDCKTZ]\d{1,4}(?![0-9A-Za-z])|[A-Z]{2}\d{3,4}(?![0-9A-Za-z])")

def _recent_choice_text(state) -> str:
    """取"用户本段所选班次"的那条发言（跳过纯确认语，如确认轮里的"确认"）。

    确认轮里最近一条用户消息是"确认"（不含车次号），若只看它就无法回填所选班次，
    段游标推不动、流程卡死在第一段。故倒序扫描用户发言取最近一个含车次号的；
    遇到上一条"段已确认"的工具消息即停止——保证不会把**上一段**的车次回填给当前段
    （用户用"第一个"这类无车次的说法选择时会返回空串，交由调用方拒绝并重新展示）
    """
    try:
        for m in reversed((state or {}).get("messages") or []):
            kind = getattr(m, "type", None)
            content = str(getattr(m, "content", "") or "")
            if kind == "tool":
                # ★ 遇到"段已确认"就停：绝不把上一段的车次回填给当前段
                if getattr(m, "name", "") == "select_transport_tool" or "段已确认" in content:
                    return ""
                continue
            if kind != "human":
                continue
            if _VEHICLE_CODE_RE.search(content):
                return content
    except Exception:
        pass
    return ""
