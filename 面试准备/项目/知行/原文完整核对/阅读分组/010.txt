"""
天气服务 MCP Server
使用高德天气 API 查询未来 4 天天气预报
"""
import os
import json
import httpx
from dotenv import load_dotenv
from fastmcp import FastMCP

load_dotenv()

# ============== FastMCP 实例 ==============
mcp = FastMCP("weather-service")

# ============== 高德 API 配置 ==============
AMAP_API_KEY = os.getenv("AMAP_API_KEY")
AMAP_WEATHER_URL = "https://restapi.amap.com/v3/weather/weatherInfo"
AMAP_GEO_URL = "https://restapi.amap.com/v3/geocode/geo"  # 城市名 → adcode

# ============== 辅助函数 ==============

async def _resolve_adcode(client: httpx.AsyncClient, city: str) -> str | None:
    """高德地理编码：城市名 → adcode（行政区划编码）；解析失败返回 None"""
    res = await client.get(
        AMAP_GEO_URL,
        params={"key": AMAP_API_KEY, "address": city}
    )
    data = res.json()
    if data.get("status") == "1" and data.get("geocodes"):
        return data["geocodes"][0].get("adcode")
    return None

# ============== 工具定义 ==============

@mcp.tool()
async def get_weather_forecast(city: str, date: str | None = None) -> str:
    """
    查询城市未来天气预报。

    Args:
        city: 中文城市名，如 "福州"、"北京"、"杭州"；
              区分模糊地区建议用"城市+区县"（如 "杭州西湖区"）。
        date: 可选，指定日期 "YYYY-MM-DD"（仅支持未来 4 天内）；
              不传则返回全部预报（约 4 天）。

    Returns:
        JSON 字符串，包含 city / province / reporttime / casts 数组
        casts 里每一项：{ date, week, dayweather, nightweather,
                        daytemp, nighttemp, daywind, ... }
    """
    if not AMAP_API_KEY:
        return json.dumps({"error": "未配置 AMAP_API_KEY"}, ensure_ascii=False)

    async with httpx.AsyncClient(timeout=10.0) as client:
        try:
            # ---------- 第 1 步：城市名 → adcode ----------
            adcode = await _resolve_adcode(client, city)
            if not adcode:
                return json.dumps(
                    {"error": f"无法解析城市: {city}，请用更规范的名称（如 '福州市'）"},
                    ensure_ascii=False,
                )

            # ---------- 第 2 步：拿 adcode 查天气 ----------
            response = await client.get(
                AMAP_WEATHER_URL,
                params={
                    "key": AMAP_API_KEY,
                    "city": adcode,
                    "extensions": "all",   # "all" = 预报天气；"base" = 实况
                    "output": "JSON",
                }
            )
            data = response.json()

            if data.get("status") != "1":
                return json.dumps({
                    "error": data.get("info", "查询失败"),
                    "infocode": data.get("infocode"),
                }, ensure_ascii=False)

            forecasts = data.get("forecasts", [])
            if not forecasts:
                return json.dumps({"error": "未找到天气数据"}, ensure_ascii=False)

            forecast = forecasts[0]
            casts = forecast.get("casts", [])

            # ---------- 第 3 步：如果指定日期，就筛出那一天 ----------
            if date:
                matched = [c for c in casts if c.get("date") == date]
                casts = matched if matched else casts

            result = {
                "city": forecast.get("city"),
                "adcode": forecast.get("adcode"),
                "province": forecast.get("province"),
                "reporttime": forecast.get("reporttime"),
                "casts": casts,
            }
            return json.dumps(result, ensure_ascii=False, indent=2)

        except httpx.TimeoutException:
            return json.dumps({"error": "请求超时"}, ensure_ascii=False)
        except Exception as e:
            return json.dumps({"error": str(e)}, ensure_ascii=False)

if __name__ == "__main__":
    # stdio 传输：Server 从 stdin 读请求、往 stdout 写响应
    # 由 MCP Client 起一个子进程运行本文件
    mcp.run(transport="stdio")

"""
搜索服务 MCP Server
使用 Tavily API 搜索旅游相关信息
"""
import os
import json
from dotenv import load_dotenv
import httpx
from fastmcp import FastMCP

load_dotenv()

mcp = FastMCP("search-service")

TAVILY_API_KEY = os.getenv("TAVILY_API_KEY")
TAVILY_URL = "https://api.tavily.com/search"

@mcp.tool()
async def search_travel_info(query: str, max_results: int = 5) -> str:
    """
    搜索旅游相关信息（景点、攻略、美食、住宿等）。

    Args:
        query: 搜索关键词，如 "西安必去景点"、"成都美食推荐"、
               "北京交通指南"、"杭州住宿攻略" 等
        max_results: 返回结果数量，默认 5 条，最多 10 条

    Returns:
        JSON 字符串，包含：
        - answer: Tavily 生成的问题摘要（相当于"直接的回答"）
        - results: 详细结果列表（title / url / content）
    """
    if not TAVILY_API_KEY:
        return json.dumps({"error": "未配置 TAVILY_API_KEY"}, ensure_ascii=False)

    max_results = min(max_results, 10)

    async with httpx.AsyncClient(timeout=30.0) as client:
        try:
            response = await client.post(
                TAVILY_URL,
                json={
                    "api_key": TAVILY_API_KEY,
                    "query": query,
                    "search_depth": "basic",       # basic / advanced 两种深度
                    "max_results": max_results,
                    "include_answer": True,        # 让 Tavily 直接给摘要
                },
            )

            if response.status_code != 200:
                return json.dumps(
                    {"error": f"API 请求失败: {response.status_code}"},
                    ensure_ascii=False,
                )

            data = response.json()
            result = {
                "query": query,
                "answer": data.get("answer"),
                "results": [
                    {
                        "title": r.get("title"),
                        "url": r.get("url"),
                        "content": r.get("content", "")[:300],  # 截断，防泄漏
                    }
                    for r in data.get("results", [])
                ],
            }
            return json.dumps(result, ensure_ascii=False, indent=2)

        except httpx.TimeoutException:
            return json.dumps({"error": "请求超时"}, ensure_ascii=False)
        except Exception as e:
            return json.dumps({"error": str(e)}, ensure_ascii=False)

if __name__ == "__main__":
    mcp.run(transport="stdio")

"""
MCP 客户端管理器
统一管理所有 MCP 服务连接
"""
import asyncio
import os
import sys
import threading
import time  # 并行验证：MCP 工具调用耗时统计
from typing import Optional, List
from dotenv import load_dotenv

#   Windows 双循环兼容：
#   主循环=Selector（psycopg 异步连接池硬性要求，connection_async.py 会拒绝 Proactor），
#   但 Selector 在 Windows 上不支持子进程（NotImplementedError），而 stdio MCP 依赖子进程。
#   故 MCP 会话全部运行在专用 Proactor 后台线程循环上，工具调用跨线程调度。
_WIN = sys.platform == "win32"
_mcp_loop: Optional[asyncio.AbstractEventLoop] = None
_mcp_loop_lock = threading.Lock()

def _get_mcp_loop() -> asyncio.AbstractEventLoop:
    """惰性创建 MCP 专用 Proactor 事件循环（后台守护线程常驻运行）"""
    global _mcp_loop
    if _WIN and (_mcp_loop is None or _mcp_loop.is_closed()):
        with _mcp_loop_lock:
            if _mcp_loop is None or _mcp_loop.is_closed():
                _mcp_loop = asyncio.ProactorEventLoop()
                threading.Thread(
                    target=_mcp_loop.run_forever, daemon=True, name="mcp-proactor-loop"
                ).start()
    return _mcp_loop

async def _run_on_mcp_loop(coro):
    """把 MCP 相关协程调度到 Proactor 线程循环执行；非 Windows 原地执行"""
    if not _WIN:
        return await coro
    return await asyncio.wrap_future(
        asyncio.run_coroutine_threadsafe(coro, _get_mcp_loop())
    )

    async def _load_tools(self) -> List:
        """ 按服务隔离加载：单个 MCP 服务故障只跳过自身，不拖垮其他服务

        原实现单个 MultiServerMCPClient 把 9 个连接放在一个 TaskGroup 里，
        任一远程主机抖动（间歇性 ConnectError）即全部报废。
        现改为每服务独立客户端 + gather 收集，失败服务记日志返回空列表。
        """
        async def load_one(name: str, client: MultiServerMCPClient) -> List:
            #  失败自动重试一次：远程 MCP 握手受网络抖动影响概率性失败
            # （实测 aigohotel-mcp 连续两次启动失败、单测直连即成功），重试可消除偶发抖动
            for attempt in (1, 2):
                try:
                    return await _run_on_mcp_loop(client.get_tools())
                except Exception as e:
                    if attempt == 1:
                        app_logger.warning(f"⚠️ MCP 服务 {name} 加载失败，2 秒后重试: {e}")
                        await asyncio.sleep(2)
                    else:
                        app_logger.warning(f"⚠️ MCP 服务 {name} 重试仍失败（本轮跳过）: {e}")
                        return []

        clients = {n: MultiServerMCPClient({n: cfg}) for n, cfg in self._configs.items()}
        results = await asyncio.gather(*(load_one(n, c) for n, c in clients.items()))
        self._clients = clients
        tools = [t for group in results for t in group]
        if not tools:
            raise RuntimeError("所有 MCP 服务均不可用")
        app_logger.info(f"✅ MCP 工具隔离加载完成: {len(tools)} 个（"
                        f"{sum(1 for r in results if r)}/{len(clients)} 服务在线）")
        return tools

    async def initialize(self, servers: List[str] = None):
        """
        初始化 MCP 客户端

        Args:
            servers: 要启用的服务列表，默认启用所有
        """
        if getattr(self, "_configs", None) is not None:
            app_logger.warning("⚠️ MCP 客户端已初始化，跳过")
            return

        # 默认启用所有服务
        servers = servers or list(self.SERVER_CONFIGS.keys())
        configs = {k: v for k, v in self.SERVER_CONFIGS.items() if k in servers}

        app_logger.info(f"初始化 MCP: {list(configs.keys())}")
        self._configs = configs

        # 预加载工具 ：在 Proactor 线程循环上建立会话（stdio 子进程依赖），按服务隔离加载
        try:
            self._tools = self._wrap_hopping(await self._load_tools())
        except Exception as e:
            app_logger.warning(f"⚠️ 预加载工具失败: {e}")
            self._tools = []

    @staticmethod
    def _wrap_hopping(tools: List):
        """ 为每个 MCP 工具的协程包一层跨循环调度（会话绑定在 Proactor 线程循环上）"""
        for t in tools:
            original = getattr(t, "coroutine", None)
            if original is not None:
                async def _hopped(*args, _orig=original, _tool_name=t.name, **kwargs):
                    #  降级处理：远程服务抖动/超时/配额超限时不抛异常，
                    # 而是把失败转成"工具结果"文本喂给模型，由模型向用户转述
                    # （如"酒店信息暂无搜索到"），避免技术报错炸穿到聊天气泡，流程不中断
                    #  并行验证：记录每次调用的起点/终点耗时，
                    # 多个工具的开始时间戳重叠即可证明同轮并行执行生效
                    _t0 = time.perf_counter()
                    app_logger.info(f"⏱️ MCP 工具 {_tool_name} 开始调用")
                    last_exc: Exception = RuntimeError("unknown")
                    #  并发容错：运行期调用失败（如 HTTP 服务被并行请求炸坏会话）先延迟重试一次
                    for attempt in (1, 2):
                        try:
                            result = await _run_on_mcp_loop(_orig(*args, **kwargs))
                            suffix = "（第2次重试成功）" if attempt == 2 else ""
                            app_logger.info(
                                f"⏱️ MCP 工具 {_tool_name} 调用完成，耗时 {time.perf_counter() - _t0:.2f}s{suffix}"
                            )
                            return result
                        except Exception as e:
                            last_exc = e
                            app_logger.warning(
                                f"MCP 工具 {_tool_name} 第 {attempt} 次调用失败（{time.perf_counter() - _t0:.2f}s）: {e}"
                            )
                            if attempt == 1:
                                await asyncio.sleep(1.5)  # 给服务喘息窗口后重试
                    #  自愈：重试仍失败说明连接已被破坏，重置连接缓存，
                    # 下一轮 middleware 重新加载步骤工具时按配置全量重建（免重启后端）
                    MCPClientManager.invalidate()
                    app_logger.warning(
                        f"MCP 工具 {_tool_name} 重试仍失败（{time.perf_counter() - _t0:.2f}s），已重置连接缓存并降级返回"
                    )
                    return (
                        (
                            f"【外部服务暂时不可用】{_tool_name} 本次调用失败（{type(last_exc).__name__}）。"
                            f"请向用户说明该信息暂无搜索到，建议稍后重试，或引导用户跳过此步骤继续行程规划；"
                            f"禁止向用户暴露技术报错细节。"
                        ),
                        None,
                    )
                t.coroutine = _hopped
        return tools

    @classmethod
    def invalidate(cls):
        """连接自愈：运行期调用炸坏会话（如 HTTP 服务被并发请求打出 TaskGroup 异常、
        连接被服务端断开）时，清空客户端与工具缓存（保留 _configs 初始化标志）；
        下次 get_tools() 会按配置全量重建连接，免重启后端即可恢复服务可用。
        """
        cls._clients = None
        cls._tools = None
        if cls._instance is not None:
            cls._instance._clients = None
            cls._instance._tools = None
        app_logger.warning("♻️ MCP 连接缓存已重置（调用重试仍失败），下次获取工具时将重建全部服务连接")

    async def close(self):
        """关闭客户端"""
        if getattr(self, "_configs", None) is not None:
            self._configs = None
            self._clients = None
            self._tools = None
            app_logger.info("MCP 客户端已关闭")

    #充当对外接口(懒加载策略)
    async def get_tools(self) -> List:
        """
        获取所有 MCP 工具

        Returns:
            LangChain 工具列表
        """
        if getattr(self, "_configs", None) is None:
            raise RuntimeError("MCP 客户端未初始化，请先调用 initialize()")

        # 如果已缓存，直接返回
        if self._tools:
            return self._tools

        # 否则重新获取 ：同样在 MCP 循环上建立会话并包装跨循环调度（带重试）
        self._tools = self._wrap_hopping(await self._load_tools())
        return self._tools

async def get_mcp_client(servers: List[str] = None) -> MCPClientManager:
    """获取 MCP 客户端管理器实例"""
    return await MCPClientManager.get_instance(servers)

# _hopped 内部用 _run_on_mcp_loop 调用原协程
