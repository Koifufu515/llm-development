async def apply_driving_selection(
    conversation_id: str, payload: DrivingSubmitPayload
) -> DrivingSubmitResult:
    """前端一次性提交 → 校验 + 写段账本 + 推进到住宿步（全程只读确认，不做任何校验）"""
    result = validate_driving_submission(payload, cache_key=conversation_id)
    if not result.ok:
        return result
    agent, config, state = await get_agent_and_state(conversation_id)   # ← 复用 base
    # 步骤守卫（代码层，不看模型措辞）：只有仍停在交通步才允许落库推进——
    # 重复提交（前端连点/已确认过）按幂等成功处理，避免把已推进的状态机拉回来
    step = str(state.get("current_step") or "")
    if step != "transport_planning":
        if step in _AFTER_TRANSPORT:
            return DrivingSubmitResult(
                ok=True, code="OK", msg="该批自驾路线已确认，无需重复提交",
                accepted=list(payload.segments),
                texts=list(state.get("selected_transport_legs") or []),
            )
        return DrivingSubmitResult(
            ok=False, code="STEP_MISMATCH",
            msg=f"当前流程步骤为 {step or '未知'}，不在交通确认阶段，无法提交自驾路线",
        )
    legs = _legs_with_dates(state)
    cached = get_cached_payload(conversation_id)
    by_id = {seg.segment_id: seg for seg in (cached.segments if cached else [])}
    selections: dict[str, str] = {}
    texts: list[str] = []
    for index, leg in enumerate(legs):
        seg = by_id.get(f"s{index + 1}")
        key = f"{leg.from_}→{leg.to}"
        if seg is not None and seg.ok:
            selections[key] = _fmt_route(seg)
            texts.append(f"{key} {_fmt_route(seg)}")
        elif seg is not None and not seg.ok:
            # ★ 该段实时无路线（高德查询失败、卡片里置灰）：占位放行，让用户能走下去
            selections[key] = _LEG_UNAVAILABLE
            texts.append(f"{key} {_LEG_UNAVAILABLE}")

            ok=False, code="INCOMPLETE",
            msg=f"提交的自驾路线未能覆盖全部行程段（应 {len(legs)} 段，可确认 {len(selections)} 段），"

        "leg_cursor": len(legs),
        "selected_transport": "driving",
        "selected_transport_legs": texts,

    })
    app_logger.info(f"✅ 多段自驾已提交并落库：{len(selections)} 段，推进到住宿步")

        ok=True, code="OK",

def _to_number(value: Any) -> float:
    """从任意形态取值里抠出数字（'¥302' / '302元' / '4.5分' / 302 → 302.0）；取不到返回 0"""
    if isinstance(value, (int, float)):
        return float(value)
    hit = re.search(r"\d+(?:\.\d+)?", str(value or ""))
    return float(hit.group()) if hit else 0.0

def _pick(record: dict, keys: tuple[str, ...]) -> Any:
    """按候选键名取值（键名大小写/下划线不敏感）；取不到返回 None"""
    # 键名归一：去下划线 + 去空白 + 转小写（纯字符串处理，等价于原 re.sub(r"[_\s]", "", k)）
    flat = {"".join(str(k).replace("_", "").split()).lower(): v for k, v in record.items()}
    for key in keys:
        value = flat.get(key)
        if value not in (None, "", [], {}):
            return value
    return None

def _as_text(value: Any) -> str:
    """任意取值转文本"""
    return str(value).strip()

def _numeric(record: dict, keys: tuple[str, ...]) -> float:
    """按候选键取**标量**数值（跳过 dict/list，避免把嵌套 price 对象误当价格）"""
    # 键名归一：去下划线 + 去空白 + 转小写（与 _pick 同款纯字符串处理）
    flat = {"".join(str(k).replace("_", "").split()).lower(): v for k, v in record.items()}
    for key in keys:
        value = flat.get(key)
        if isinstance(value, (int, float, str)) and _to_number(value):
            return _to_number(value)
    return 0.0

def _price_pair(record: dict, nights: int) -> tuple[float, float]:
    """取（总价, 每晚单价）：嵌套 price 对象 → 扁平键，缺一方按晚数互补

    实测 aigohotel 的 price 是对象且含 lowestPrice（每晚价），故：
    lowestPrice 当每晚价；总价不返回时按 每晚价 × 晚数 换算（纯单位换算，不做业务校验）。
    """
    total = night = 0.0
    nested = next((v for v in (_pick(record, (k,)) for k in _PRICE_OBJ_KEYS) if isinstance(v, dict)), None)
    if nested:
        night = _numeric(nested, _PRICE_NIGHT_KEYS)
        total = _numeric(nested, _PRICE_TOTAL_KEYS)
    total, night = total or _numeric(record, _TOTAL_KEYS), night or _numeric(record, _NIGHT_KEYS)
    if not total and night:
        total = night * nights
    elif total and not night:
        night = total / nights
    return total, night
