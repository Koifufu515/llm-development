class WeatherCast(BaseModel):
    """单日预报（字段与高德 casts 对齐；缺字段给默认空串，不报错、渲染时留空）"""
    date: str = ""
    dayweather: str = ""
    nightweather: str = ""
    daytemp: str = ""
    nighttemp: str = ""
    daywind: str = ""
    daypower: str = ""

class WeatherForecast(BaseModel):
    """天气预报整体结构（第 12 章流式渲染天气会 import 复用它）"""
    city: str = ""
    reporttime: str = ""
    casts: list[WeatherCast] = Field(default_factory=list)

def _normalize_weather_payload(raw) -> dict:
    """★ 归一化漏斗：把 MCP 三种返回形态统一成 dict（第 12 章也会 import 复用）

    历史 bug：MCP adapter 返回的常是"内容块列表" [{'type':'text','text':'{json}'}]，
    旧代码只按 str/dict 解析 → list 分支没接住 → 恒判"天气为空" → 每次都降级。
    接口本来是好的，纯粹解析口径错，所以先在这里把形态抹平。
    """
    if isinstance(raw, list):                   # 形态一：内容块列表 → 把每块的 text 拼起来
        texts = [str(b.get("text") or "") if isinstance(b, dict) else str(getattr(b, "text", "") or "")
                 for b in raw]
        raw = "\n".join(t for t in texts if t).strip()
    if isinstance(raw, str):                     # 形态二：JSON 字符串 → 抠出 {...} 再 loads
        text = raw.strip()
        if not text.startswith("{"):             # 兼容"说明文字 + JSON"混合文本
            start, end = text.find("{"), text.rfind("}")
            text = text[start:end + 1] if 0 <= start < end else text
        raw = json.loads(text)
    return raw if isinstance(raw, dict) else {}  # 形态三：本来就是 dict → 直接用

async def weather_agent_node(state: dict) -> dict:
    """天气 Agent 节点：调真实高德天气；查不到就如实说，绝不编造"""
    destination = state["destination"]
    app_logger.info(f"🌤️ 天气 Agent 执行: {destination}")

    report = f"## {destination} 天气信息\n\n"
    try:
        from app.tools.mcp_tools import get_weather_tools   # 延迟导入，与工具装载解耦
        tools = await get_weather_tools()
        tool = next((t for t in tools if t.name == "get_weather_forecast"), None)
        if tool is None:
            raise RuntimeError("天气工具未加载")

        # 不传 date：该参数只支持未来 4 天内，超出会返回空预报
        payload = _normalize_weather_payload(await tool.ainvoke({"city": destination}))
        if payload.get("error"):
            raise RuntimeError(str(payload["error"]))

        forecast = WeatherForecast.model_validate(payload)   # Pydantic 校验 + 填默认
        if not forecast.casts:
            raise RuntimeError("天气返回为空")

        lines = [f"📅 {c.date}：白天{c.dayweather} {c.nighttemp}~{c.daytemp}°C，"
                 f"{c.daywind}风 {c.daypower} 级" for c in forecast.casts]
        report += "\n".join(lines)
        report += f"\n\n*数据来源：高德天气（发布于 {forecast.reporttime}）*"
    except Exception as e:
        # ★ 铁律：查不到宁可说"未获取到"，也绝不编造天气（编造会被写进行程正文误导用户）
        app_logger.warning(f"天气查询失败（降级不编造）: {destination} | {type(e).__name__}: {e}")
        report += "暂未获取到实时天气，请以当地实际天气为准。"

    return {"agent_results": [{"agent_name": "weather", "result": report}]}

async def synthesizer_node(state: DestinationRouterState) -> dict:
    """综合器节点（扇入）：等所有并行分支跑完，把结果拼成一份报告"""
    app_logger.info("综合 Agent 结果...")
    results = state["agent_results"]           # reducer 已把各分支结果合并到这里
    if not results:
        return {"final_report": "未找到相关信息。"}
    # 简单拼接就够：攻略段 + 天气段各自成块，顺序无所谓（也可换 LLM 做更智能的综合）
    final_report = "\n\n".join(r["result"] for r in results)
    app_logger.info("✅ 综合完成")
    return {"final_report": final_report}

def create_destination_router():
    """组装并编译整张子图 —— 手工建图四步：建图 → 加节点 → 连边 → compile"""
    workflow = StateGraph(DestinationRouterState)   # ① 以 State 为骨架建图

    # ② 加节点：名字 → 处理函数（名字就是 Send / 边里用的"节点名"）
    workflow.add_node("classifier", classifier_node)
    workflow.add_node("explore", explore_agent_node)
    workflow.add_node("weather", weather_agent_node)
    workflow.add_node("synthesizer", synthesizer_node)

    # ③ 连边
    workflow.add_edge(START, "classifier")          # 入口 → 分类器
    workflow.add_conditional_edges(                 # 分类器 → 条件边（route_to_agents 里用 Send 扇出）
        "classifier", route_to_agents, ["explore", "weather"],  # 第三参：声明可能的目标节点
    )
    workflow.add_edge("explore", "synthesizer")     # 两个分支都汇到综合器（扇入）
    workflow.add_edge("weather", "synthesizer")
    workflow.add_edge("synthesizer", END)           # 综合器 → 出口

    app = workflow.compile()                        # ④ 编译成可 ainvoke 的子图
    app_logger.info("✅ 目的地 Router 创建完成")
    return app

"""快速通道服务。本章先放四个路段助手（拆多城、排游玩顺序、造交通段、分配住宿晚数），第 12 章整份重写本文件"""
import re

# 目的地支持逗号分隔多城（"杭州,深圳"），全链路真查每段交通 + 每城酒店/天气。
# 串联语义：出发地 → city0 → city1 → ... → 返程（返程终点=出发地）。
# 原本把整串"杭州,深圳"当单个城市塞进所有查询（toStation/arrcity/place/city），
# 导致交通全查空 + LLM 自行扩天数为 5 天。
_CITY_DELIMS = "，,、"

def _split_cities(destination: str) -> list[str]:
    """把 destination 按中/英逗号、顿号切分为城市列表（去空去重保序）"""
    if not destination:
        return []
    flat = destination
    for ch in _CITY_DELIMS:
        flat = flat.replace(ch, " ")
    out: list[str] = []
    for city in flat.split():
        if city not in out:
            out.append(city)
    return out

def ordered_play_cities(departure: str, destination: str) -> list[str]:
    """游玩/住宿节点城市序列（去重保序）：**出发地仅当本身也在目的地串里才计入**，且置于最前。

 （用户规格）：住宿站点 = 目的地城市序列。北京→上海 = [上海]（不再多查北京），
    北京→天津/上海 = [天津,上海]，北京→北京 = [北京]；目的地含出发地（北京→北京/天津）时
    出发地是"第一游玩地"（先本地玩再走），故置于最前 → [北京,天津]
    """
    cities = _split_cities(destination)
    dep = (departure or "").strip()
    # ★ 出发地也在目的地里：先在出发地玩
    ordered = (([dep] if dep and dep in cities else []) + cities)
    out: list[str] = []
    for c in ordered:
        if c and c not in out:
            out.append(c)
    return out

def _build_legs(departure: str, cities: list[str]) -> list[tuple[str, str]]:
    """构造交通路段序列：出发→首城、城际转场、末城→出发地返程。

    剔除"自身段"——当目的地含出发地（如出发+目的地都是北京）时，
    route 会出现"北京→北京"，本无意义；剔除后行程正确保留北京本地游玩。
    节点列表 = 出发地 + 目的地序列 + 出发地（返程），相邻节点城市不同才生成段：
    北京→上海 = 2 段、北京→天津→上海 = 3 段、北京→北京 = 0 段；出发地缺失时不补返程
    """
    if not cities:
        return []
    route = [departure] + cities + ([departure] if departure else [])
    # ★ 相邻打包成段；a == b 的自身段跳过；末尾补回的出发地让最后一段成为返程
    return [(a, b) for a, b in zip(route[:-1], route[1:]) if a != b]

def _stay_nights(days: int, stations: int) -> list[int]:
    """住宿晚数分配：**出发地固定 1 晚**，剩余天数在其后城市均分（前几城多担 1 天）

 （用户规格）：酒店段覆盖全部节点城市（含出发地）。天数不够分时
    每站仍按 1 晚顺排（不替用户删城），与 _stay_windows 原口径一致
    """
    if stations <= 0 or days <= 0:
        return []
    if stations == 1:
        return [days]
    rest = max(1, days - 1)
    per, rem = divmod(rest, stations - 1)
    # ★ 首站 1 晚，剩下的天数在后面几站均分，除不尽的前几站多担
    return [1] + [per + (1 if i < rem else 0) for i in range(stations - 1)]

def _leg_query_date(state: dict, legs: list, cursor: int) -> str:
    """ 多段行程的"段日期"按时序分配（代码算，不由模型猜）

    实测多城会话三段城际全按出发日查询（重庆→北京、北京→天津都查 2026-09-18），
    与实际时序矛盾（人不可能同一天既在重庆又在北京又在天津）。规则：
      ① 出发地→首城 = 出发日；
      ② 城市 i-1 → 城市 i = 出发日 + 前 i 站累计晚数（即下一城的首日）；
      ③ 返程段（末段、终点=出发地）= 末日。
    晚数分配与酒店侧同源：出发地 1 晚 + 剩余天数在其后城市均分
    """
    from datetime import date as _date, timedelta as _timedelta
    req = state.get("user_requirement") or {}
    days = int(req.get("travel_days") or 0)
    try:
        start = _date.fromisoformat(str(req.get("departure_date") or "")[:10])
    except Exception:
        return ""
    if days <= 0 or not legs:
        return ""
    c = max(0, min(int(cursor), len(legs) - 1))
    from app.services.express_planning_service import _stay_nights, ordered_play_cities
    # 与住宿站点同源对齐——节点 = ordered_play_cities（目的地城市串，
    # 出发地仅当本身也在目的地串里才计入并置最前），晚数按**真实节点数**分配（此前按
    # legs+1 估，3 城以上与酒店侧错位）。出发地本身是游玩首节点时（北京→北京/天津），
    # 首段（北京→天津）发生在"北京住 1 晚之后"，偏移整体 +1 站
    dep = str(req.get("departure_city") or "").strip()
    dest = str(state.get("selected_destination") or "") or str(req.get("destination") or "")
    # ★ 返程段 = 末日：单独判定，不靠最后一行 min(offset, days - 1) 的截断碰巧落到末日
    if dep and c == len(legs) - 1 and str(legs[c]).rpartition("→")[2].strip() == dep:
        return (start + _timedelta(days=days - 1)).isoformat()
    nodes = ordered_play_cities(dep, dest)
    # ★ 与住宿步同一口径：首站 1 晚、其余均分
    nights = _stay_nights(days, len(nodes) or (len(legs) + 1))
    # ★ 出发地也在目的地里：第一站在本地玩，每段都往后挪一站
    shift = 1 if (dep and dep in nodes and nodes[0] == dep) else 0
    offset = sum(nights[:min(c + shift, len(nights))]) if nights else 0
    return (start + _timedelta(days=min(offset, days - 1))).isoformat()
