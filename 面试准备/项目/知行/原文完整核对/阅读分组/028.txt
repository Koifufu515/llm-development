async def auto_select_hotels(conversation_id: str, state: dict) -> list[str]:
    """自动确认模式：不弹卡片，由代码代选全部站并推进（门店取首条 / 品牌取首个）

    返回已选定站的展示文本（空列表 = 未推进，调用方跳过即可）
    """
    segments = _segments(state)
    if not segments:
        return []
    payload = await _payload_for(conversation_id, segments)
    texts: list[str] = []
    prices: dict[str, float] = {}
    for seg in payload.segments:
        head = f"{seg.city}（{seg.checkInDate} 至 {seg.checkOutDate}）"
        if seg.hotel_list:
            hotel = seg.hotel_list[0]
            texts.append(
                f"{head}：{hotel.hotelName}，{hotel.pricePerNight:g}元/晚，总价 {hotel.totalPrice:g}元"
            )
            if float(hotel.pricePerNight) > 0:     # ★ 同一次循环产出数字账本
                prices[seg.city] = float(hotel.pricePerNight)
        elif seg.brand_list:
            texts.append(
                f"{head}：{seg.brand_list[0].brandName}（实时报价不可用，仅选定品牌，"
                f"具体门店到店择优选同档位）"
            )
    if len(texts) != len(payload.segments):
        app_logger.warning(
            f"⚠️ 住宿自动模式代选不完整（{len(texts)}/{len(payload.segments)} 站有候选），本轮不推进"
        )
        return []
    agent, config, _ = await get_agent_and_state(conversation_id)
    extra, step = _next_step(state)
    upd: dict = {
        "selected_hotels": texts, "stay_cursor": len(texts), "current_step": step,
        "stay_prices": prices,      # ★ 数字账本：第 11 章预算工具直接读，不解析展示文本
        # 档位必须**无条件写入**（表单未预选时兜底经济型）。
        # 行程步 requires 硬校验 selected_accommodation_types，字段缺失会直接抛错卡死流程
        "selected_accommodation_types": _accom_types(state) or ["economy_hotel"],
        **extra,
    }
    await agent.aupdate_state(config, upd)
    app_logger.info(f"✅ 住宿自动模式已代选 {len(texts)} 站，推进到 {step}")
    return texts

async def apply_hotel_selection(
    conversation_id: str, payload: HotelMultiSegmentSubmitPayload
) -> HotelMultiSegmentSubmitResult:
    """前端一次性提交 → 校验 + 写住宿账本 + 推进到餐饮/行程步（**全程不做日期校验**）"""
    result = validate_hotel_multi_submission(payload, cache_key=conversation_id)
    if not result.ok:
        return result
    agent, config, state = await get_agent_and_state(conversation_id)
    # 步骤守卫（代码层，不看模型措辞）：只有仍停在住宿步才允许落库推进——
    # 重复提交（前端连点/已确认过）按幂等成功处理，避免把已推进的状态机拉回住宿步
    step = str(state.get("current_step") or "")
    if step != "accommodation_planning":
        if step in _AFTER_STAY:
            return HotelMultiSegmentSubmitResult(
                ok=True, code="OK", msg="该批住宿已确认，无需重复提交",
                accepted=list(payload.segments),
                texts=list(state.get("selected_hotels") or []),
                next_step=step,      # 幂等：反馈文案用当前实际所在步骤
            )
        return HotelMultiSegmentSubmitResult(
            ok=False, code="STEP_MISMATCH",
            msg=f"当前流程步骤为 {step or '未知'}，不在住宿确认阶段，无法提交住宿",
        )
    extra, nxt = _next_step(state)
    upd: dict = {
        "selected_hotels": result.texts, "stay_cursor": len(result.texts),
        "stay_prices": dict(result.prices),   # ★ 校验器已经算好，这里只负责落库
        "current_step": nxt,
        # 档位无条件写入（表单未预选 → 兜底经济型），
        # 否则行程步 requires 校验 selected_accommodation_types 缺失直接抛错
        "selected_accommodation_types": _accom_types(state) or ["economy_hotel"],
        **extra,
    }
    await agent.aupdate_state(config, upd)
    app_logger.info(f"🏨 多段住宿已提交并落库：{len(result.texts)} 站，推进到 {nxt}")
    return result.model_copy(update={
        "msg": f"已确认 {len(result.texts)} 站住宿，继续为您安排餐饮与行程",
        "next_step": nxt,   # 提交路由据此挑续跑反馈文案（行程步 → "正在生成行程报告"）
    })

import re
_DAY_MARK = re.compile(r"第\s*(\d+|[一二三四五六七八九十]+)\s*天|Day\s*\d", re.IGNORECASE)
def _history_has_itinerary(msgs: list) -> bool:
    """判断本回合历史里是否已输出过逐日行程正文
    行程轮规则：先展示正文+预算征询、用户满意后才落库，确认轮的必要条件是
    历史里存在 Day X 正文；若没有（例如用户第一轮直接说"满意"），不能强制落库
    """
    for m in msgs:
        if getattr(m, "type", None) != "ai":
            continue
        content = getattr(m, "content", "") or ""
        if isinstance(content, str) and _DAY_MARK.search(content):
            return True
    return False

# 残缺工具调用历史自愈。当工具节点写状态阶段抛异常时，checkpoint 会残留
# AIMessage 带 tool_calls、但无对应 ToolMessage 的残缺历史；后续传给模型会报400错误：
# InternalError.Algo.InvalidParameter: An assistant message with "tool_calls" must be followed
# by tool messages responding to each "tool_call_id"
# 会话会直接卡死，只能新建会话。每次模型调用前做幂等体检：
# ① 存在未响应的 tool_call → 紧跟该AIMessage补一条说明性ToolMessage，保证消息顺序合法；
# ② 找不到对应AIMessage的孤立ToolMessage → 丢弃，避免同类400错误。
_TOOL_ABORTED_HINT = (
    "【系统提示】该工具调用未执行完成（回合内被中断），没有可用结果，请勿据此作答。"
)
def _repair_dangling_tool_calls(msgs: list) -> tuple[list, int, int]:
    """修复悬空 tool_calls / 孤立 tool 消息
    返回 (处理后的消息列表, 补入的 ToolMessage 条数, 丢弃的孤立 ToolMessage 条数)；
    无需修复时原样返回入参对象（零拷贝，不影响正常路径）
    """
    out: list = []
    inserted = dropped = 0
    i, n = 0, len(msgs)
    while i < n:
        m = msgs[i]
        mtype = getattr(m, "type", None)
        if mtype == "ai" and getattr(m, "tool_calls", None):
            out.append(m)
            # 本AI消息声明的 tool_call_id → 工具名，已响应的逐条移除
            pending = {
                tc.get("id"): tc.get("name")
                for tc in m.tool_calls if tc.get("id")
            }
            j = i + 1
            while j < n and getattr(msgs[j], "type", None) == "tool":
                pending.pop(getattr(msgs[j], "tool_call_id", None), None)
                out.append(msgs[j])
                j += 1
            for tid, tname in pending.items():
                # 补缺失响应：追加中断提示ToolMessage，不删除原AI消息
                out.append(ToolMessage(
                    content=_TOOL_ABORTED_HINT, tool_call_id=tid, name=tname or "tool"
                ))
                inserted += 1
            i = j
            continue
        # 前面没有对应 tool_calls 的孤立 tool 消息：直接丢弃
        if mtype == "tool":
            dropped += 1
            i += 1
            continue
        out.append(m)
        i += 1
    if inserted or dropped:
        return out, inserted, dropped
    return msgs, 0, 0

# ============== 上下文窗口兜底（按"用户轮"边界丢弃最旧历史） ==============
# TravelState.messages 采用追加式存储，每次调用模型都会全量传入历史；
# 入口层已对工具结果做确定性裁剪，但多轮累积仍会触发超长输入超限。窗口规则：
# ① 仅裁剪发给模型的请求视图，checkpoint内原始历史完整保留；
# ② 切点落在HumanMessage，整轮丢弃，禁止拆分AIMessage与ToolMessage配对，防止制造残缺历史；
# ③ 当前轮（最后一条HumanMessage起）永久保留，回合内工具往返上下文不丢失。
_WINDOW_TRIGGER = 40   # 消息总数超过阈值才触发裁剪，普通会话不会触发
_WINDOW_KEEP = 24      # 目标保留消息下限；切点向前寻找用户轮边界，保留数量只会更多
def _trim_window(msgs: list) -> list:
    """把最旧的历史轮整轮丢弃；找不到安全切点则原样返回
    切点取目标窗口起点与当前轮起点之间最近的HumanMessage；
    """
    if len(msgs) <= _WINDOW_TRIGGER:
        return msgs
    last_human = -1
    for i in range(len(msgs) - 1, -1, -1):
        if isinstance(msgs[i], HumanMessage):
            last_human = i
            break
    if last_human <= 0:
        # 没有更早历史轮可裁剪
        return msgs
    cut = min(len(msgs) - _WINDOW_KEEP, last_human - 1)
    for i in range(cut, -1, -1):
        if isinstance(msgs[i], HumanMessage):
            kept_system = [m for m in msgs[:i] if getattr(m, "type", None) == "system"]
            return kept_system + msgs[i:]
    return msgs
