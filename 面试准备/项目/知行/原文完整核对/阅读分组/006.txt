"""
StepConfigMiddleware
根据 State.current_step 动态注入 Prompt 和 Tools。
"""
from typing import Callable
from jinja2 import Template

from langchain.agents.middleware import AgentMiddleware, ModelRequest, ModelResponse

from app.core.state import TravelState
from app.utils.logger import app_logger

class StepConfigMiddleware(AgentMiddleware):
    """
    步骤配置中间件

    职责：
      每一次 LLM 调用之前：
        1. 读 State.current_step
        2. 从 step_config 中查该步的 Prompt / Tools / Requires
        3. 前置依赖校验（Requires 的字段必须都在 State 中）
        4. 用 Jinja2 渲染 Prompt 里的 {{ var }} 占位符
        5. request.override(...) 注入 system_prompt + tools
    """

    def __init__(self, step_config: dict):
        """
        Args:
            step_config: 预加载的 StepConfig（见 app/agents/handoffs/step_config.py）
        """
        self._step_config = step_config

    async def awrap_model_call(
            self,
            request: ModelRequest,
            handler: Callable[[ModelRequest], ModelResponse]
    ) -> ModelResponse:
        # ---------- 1. 读当前步骤 ----------
        state: TravelState = request.state
        # State 支持 .get() 但为了给 Jinja2 用，转成普通 dict 更保险
        state_dict = dict(state) if hasattr(state, "items") else {}
        current_step = state.get("current_step", "requirement_collection")
        user_id = state.get("user_id")

        app_logger.info(f"📋 用户 ID: {user_id}")
        app_logger.info(f"📍 当前步骤: {current_step}")

        # ---------- 2. 查 step_config ----------
        if current_step not in self._step_config:
            app_logger.error(f"❌ 未知步骤: {current_step}")
            raise ValueError(f"未知步骤: {current_step}")

        step_config = self._step_config[current_step]

        # ---------- 3. 前置依赖校验 ----------
        for required_field in step_config["requires"]:
            if required_field not in state or state[required_field] is None:
                error_msg = (
                    f"步骤 {current_step} 需要 State 里已存在字段: {required_field}, "
                    f"但当前 State 中该字段为空"
                )
                app_logger.error(f"❌ {error_msg}")
                app_logger.error(f"   当前 State keys = {list(state.keys())}")
                raise ValueError(error_msg)

        # ---------- 4. Jinja2 渲染 Prompt ----------
        try:
            template = Template(step_config["prompt"])
            system_prompt = template.render(**state_dict)
            # ** 解包：把 dict 里每个 key 当变量名传给模板
        except Exception as e:
            app_logger.warning(f"⚠️ 提示词渲染失败: {e}，回退到原始模板")
            system_prompt = step_config["prompt"]

        # ---------- 5. 注入 request ----------
        tools = step_config["tools"]
        modified_request = request.override(
            system_prompt=system_prompt,
            tools=tools,
        )

        app_logger.info(f"✅ 已注入配置: {len(tools)} 个工具")

        # ---------- 6. 交给下一环（真的调 LLM）----------
        return await handler(modified_request)

# ============== 工厂函数 ==============

async def create_step_config_middleware() -> StepConfigMiddleware:
    """
    工厂：从 step_config 模块加载配置并构造 Middleware。
    异步，因为 get_step_config() 是异步（后续章节要往里加 await get_mcp_tools() 之类）
    """
    from app.agents.handoffs.step_config import get_step_config

    step_config = await get_step_config()
    return StepConfigMiddleware(step_config)

from app.core.checkpointer import get_checkpointer
from app.core.store import get_store

async def create_travel_agent():
    """
    创建 Handoffs 旅行规划 Agent（8 步 + Middleware 版）

    返回：编译好的 Agent

    注意：
        必须在 lifespan 内调用（Checkpointer / Store 需要事先启动）
    """
    app_logger.info("创建 Travel Agent（8 步 + Middleware）...")

    # 1. 拿大脑
    llm = get_qwen_llm(streaming=True)

    # 2. 总工具表：8 步的所有工具都注册在这里
    #    Middleware 会按 current_step 筛出该步能用的子集
    all_tools = [
        record_requirement_tool,
        select_destination_tool,
        select_transport_tool,
        select_accommodation_tool,
        select_food_tool,
        generate_itinerary_tool,
        summarize_budget_tool,
    ]

    # 3. 中间件：预加载 StepConfig
    step_config_middleware = await create_step_config_middleware()

    # 4. 拿持久化
    checkpointer = await get_checkpointer()
    store = await get_store()

    # 5. 组装
    agent = create_agent(
        model=llm,
        tools=all_tools,
        state_schema=TravelState,
        middleware=[step_config_middleware],  # ← 本章的关键变化
        checkpointer=checkpointer,
        store=store,
    )

    app_logger.info("✅ Travel Agent（8 步 + Middleware）创建完成")
    return agent

    return text[:n] + "..."

async def main():
    print("╔" + "═" * 58 + "╗")
    print("║" + " 第 04 章 · Middleware 驱动 8 步流程冒烟".center(58) + "║")
    print("╚" + "═" * 58 + "╝")

    async with checkpointer_lifespan():
        async with store_lifespan():
            agent = await create_travel_agent()

            config = {
                "configurable": {
                    "thread_id": THREAD_ID,
                    "user_id": USER_ID,
                }
            }

            print("\n开始对话，每一轮打印 current_step 变化...")

            prev_step = None
            for i, user_text in enumerate(USER_TURNS, start=1):
                print("\n" + "─" * 60)
                print(f"[第 {i} 轮]")
                print(f"  用户: {user_text}")

                # 只推新消息，历史交给 Checkpointer
                input_state = {"messages": [HumanMessage(content=user_text)]}

                try:
                    result = await agent.ainvoke(input_state, config=config)
                except Exception as e:
                    print(f"  ❌ 本轮出错: {type(e).__name__}: {e}")
                    break

                # 从 result 里挑 AI 最后一条可读回复
                last_msg = result["messages"][-1]
                if isinstance(last_msg, AIMessage):
                    text = last_msg.content if isinstance(last_msg.content, str) else str(last_msg.content)
                    print(f"  Agent: {_short(text)}")
                    if last_msg.tool_calls:
                        for tc in last_msg.tool_calls:
                            print(f"    🔧 {tc['name']}({list(tc['args'].keys())})")
                elif isinstance(last_msg, ToolMessage):
                    print(f"  🛠️  {last_msg.content[:200]}")

                # 打印 current_step 变化
                cur_step = result.get("current_step")
                if cur_step != prev_step:
                    arrow = f"{prev_step or '(未开始)'} → {cur_step}"
                    print(f"  📌 步骤推进: {arrow}")
                else:
                    print(f"  📌 步骤保持: {cur_step}")
                prev_step = cur_step

                # 到达终点直接停
                if cur_step == "report_generation":
                    # 让最后一轮把报告写出来即可
                    if i >= len(USER_TURNS):
                        pass
                    elif isinstance(last_msg, AIMessage) and "旅行报告" in str(last_msg.content):
                        break

            # 结束后再拉一次 State 看总览
            print("\n" + "=" * 60)
            print(" 最终 State 概览")
            print("=" * 60)
            state = await agent.aget_state(config)
            values = state.values
            for k in [
                "current_step", "user_requirement", "selected_destination",
                "selected_transport", "selected_accommodation_types",
                "selected_food_types", "itinerary", "budget",
            ]:
                v = values.get(k)
                if v is None:
                    print(f"  {k}: (未填)")
                else:
                    s = str(v)
                    if len(s) > 200:
                        s = s[:200] + "..."
                    print(f"  {k}: {s}")

            print("\n" + "🎉" * 20)
            print("  第 04 章冒烟完成！")
            print("🎉" * 20 + "\n")

if __name__ == "__main__":
    asyncio.run(main())
