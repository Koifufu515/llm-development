async def create_travel_agent():
    """
    创建 Handoffs 旅行规划 Agent（8 步 + 中间件 + 回退 + 记忆）
    """
    app_logger.info("创建 Travel Agent（8 步 + 回退 + 记忆）...")

    llm = get_qwen_llm(streaming=True)

    # 总工具表：状态转换 + 回退 + 记忆
    all_tools = [
        # 状态转换（第 03-04 章）
        record_requirement_tool,
        select_destination_tool,
        select_transport_tool,
        select_accommodation_tool,
        select_food_tool,
        generate_itinerary_tool,
        summarize_budget_tool,
        # 回退（本章新增）
        *ALL_ROLLBACK_TOOLS,
        # 记忆（本章新增）
        *MEMORY_TOOLS,
    ]

    step_config_middleware = await create_step_config_middleware()
    checkpointer = await get_checkpointer()
    store = await get_store()

    agent = create_agent(
        model=llm,
        tools=all_tools,
        state_schema=TravelState,
        middleware=[step_config_middleware],
        checkpointer=checkpointer,
        store=store,
    )

    app_logger.info(f"✅ Travel Agent 创建完成，注册了 {len(all_tools)} 把总工具")
    return agent

def _short(text: str, n: int = 220) -> str:
    if not isinstance(text, str):
        text = str(text)
    return text if len(text) <= n else text[:n] + "..."

def _print_turn(agent_result: dict, user_text: str, i: int):
    print("\n" + "─" * 60)
    print(f"[第 {i} 轮]")
    print(f"  用户: {user_text}")
    last_msg = agent_result["messages"][-1]
    if isinstance(last_msg, AIMessage):
        print(f"  Agent: {_short(last_msg.content)}")
        for tc in (last_msg.tool_calls or []):
            print(f"    🔧 {tc['name']}({list(tc['args'].keys())})")
    elif isinstance(last_msg, ToolMessage):
        print(f"  🛠️  {_short(last_msg.content)}")
    print(f"  📌 current_step = {agent_result.get('current_step')}")
