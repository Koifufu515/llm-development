"""证据与引用：平台交付给上游的数据形状。"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import date
from uuid import UUID

@dataclass
class Citation:
    """一条引用。上游用它渲染来源卡片、跳转原文。"""

    document_id: UUID
    source_uri: str
    title: str | None
    url: str | None                  # 原文链接，来自 documents.meta（第三章的抓取清单写入）
    license: str | None
    heading_path: list[str]
    char_start: int                  # 命中块在原文里的起止位置
    char_end: int
    locator_kind: str                # "char" 精确到字符 | "paragraph" 只到段落 | "unknown"
    verified_at: date | None = None  # 核实日期，第六章事实层填写，本章一律为 None

@dataclass
class Evidence:
    """一条证据 = 命中块 + 上下文 + 引用 + 分数。"""

    chunk_id: str
    text: str                        # 命中块原文，引用区间指向它，用来高亮
    context: str                     # 交给大模型读的上下文
    context_level: str               # "neighbors" 带邻块 | "section" 本节 | "chunk" 只有命中块
    citation: Citation

    score: float                     # 融合分（第四章）
    rerank_score: float | None
    dense_score: float | None
    sparse_score: float | None
    dense_rank: int | None
    sparse_rank: int | None
    sibling_hits: int                # 同一小节里被合并掉的其他命中数
